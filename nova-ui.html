02b — Reactive Glass Buttons

Reactive Glassy Glow (Cursor-Following)

Fluid glass buttons with a reactive inner glow that follows your cursor inside the button. Smooth, minimal, and GPU-friendly.

Core Reactive Glass

Hover to see the glow track to your cursor position.

3D Tilt + Reactive Glow

Combine .tilt with reactive glass for a fluid micro-interaction.

Usage

Add class="btn btn--glass" and data-glow, then include the small JS below to track cursor. 

<button class="btn btn--glass ripple tilt" data-glow>
  <span>Reactive Glass</span>
</button>

<script>
  document.querySelectorAll('[data-glow]').forEach((btn)=>{
    btn.addEventListener('pointermove', (e)=>{
      const r = btn.getBoundingClientRect();
      const x = ((e.clientX - r.left) / r.width) * 100;
      const y = ((e.clientY - r.top) / r.height) * 100;
      btn.style.setProperty('--glow-x', x + '%');
      btn.style.setProperty('--glow-y', y + '%');
    }, { passive: true });
    btn.addEventListener('pointerleave', ()=>{
      btn.style.setProperty('--glow-x', '50%');
      btn.style.setProperty('--glow-y', '0%');
    });
  });
</script>

06 — Cursor

Custom Cursor (Reference)

Pulled from falsecloudd-preview.html, glow removed. Add the HTML, CSS, and JS to enable the minimal dot cursor. Respects prefers-reduced-motion and only runs on hover-capable devices.

HTML

Place this near the closing </body> tag.

<div class="cur dot" id="cd" aria-hidden="true">
  <b><i></i></b>
</div>

CSS (minimal, no glow)

Includes --text and base styles.

.cur {
  position: fixed;
  top: 0; left: 0;
  pointer-events: none;
  z-index: 9999;
  opacity: 0;
  transition: opacity .2s;
}
.cursor-on .cur { opacity: 1; }
.cursor-out .cur { opacity: 0; }
.cur.dot { width: 8px; height: 8px; }
.cur.dot b { display: block; width: 100%; height: 100%; will-change: transform, filter; }
.cur.dot i {
  display: block; width: 100%; height: 100%;
  border-radius: 50%;
  background: var(--text);
  box-shadow: none;
  transition: transform .25s cubic-bezier(.2,.8,.2,1), background-color .25s, box-shadow .25s;
}
.cur.dot.hot i { transform: scale(1.7); box-shadow: none; }
.cur.dot.down i { transform: scale(.4); transition-duration: .08s; }

JS (simplified tracking)

Attaches to #cd and toggles cursor states.

(function(){
  if(!matchMedia('(hover:hover) and (pointer:fine)').matches)return;
  var root=document.documentElement,dot=document.getElementById('cd');
  var reduce=matchMedia('(prefers-reduced-motion:reduce)').matches;
  var ease=reduce?1:.45;
  var x=0,y=0,cx=0,cy=0,on=false,raf=0,last=0,s=0,ang=0,moved=0;
  function draw(t){
    var dt=Math.min(t-(last||t),50)||16.7,k=1-Math.pow(1-ease,dt/16.67);
    last=t; var mx=(x-cx)*k,my=(y-cy)*k; cx+=mx; cy+=my;
    var sp=Math.sqrt(mx*mx+my*my)*16.67/dt;
    if(sp>.5)ang=Math.atan2(my,mx);
    var idle=t-moved>80; var target=(reduce||idle)?0:Math.min(sp/26,1);
    s+=(target-s)*(idle?.5:.3);
    dot.style.transform='translate3d('+cx+'px,'+cy+'px,0) translate(-50%,-50%)';
    if(s>.01){
      dot.firstElementChild.style.transform='rotate('+ang+'rad) scale('+(1+s*2.4)+','+(1-s*.22)+')';
    }else{s=0;dot.firstElementChild.style.transform='none';}
    if(Math.abs(x-cx)>.1||Math.abs(y-cy)>.1||s>0){raf=requestAnimationFrame(draw)}else{raf=0;last=0}
  }
  document.addEventListener('mousemove',function(e){
    x=e.clientX;y=e.clientY;moved=performance.now();
    if(!on){on=true;cx=x;cy=y;root.classList.add('cursor-on');}
    if(!raf)raf=requestAnimationFrame(draw);
  });
  document.addEventListener('mouseover',function(e){dot.classList.toggle('hot',!!e.target.closest('a,button,[role="button"]'));});
  document.addEventListener('mousedown',function(){dot.classList.add('down');},true);
  window.addEventListener('mouseup',function(){dot.classList.remove('down');},true);
  root.addEventListener('mouseleave',function(){root.classList.add('cursor-out');});
  root.addEventListener('mouseenter',function(e){x=e.clientX;y=e.clientY;cx=x;cy=y;s=0;root.classList.remove('cursor-out');});
})();
